⚙️ PENGATURAN GURU & GOOGLE SHEETS

Hubungkan aplikasi ini dengan Google Spreadsheet agar setiap tugas yang dikumpulkan siswa dari HP/Laptop mana pun langsung tersimpan otomatis ke Google Sheets & Google Drive Anda!

📌 Langkah Singkat Menghubungkan Google Sheets:

  1. Buka sheets.new di browser untuk membuat Spreadsheet baru.
  2. Klik menu Extensions → Apps Script, hapus kode lama, lalu tempel kode di bawah.
  3. Klik tombol biru Deploy → New deployment → Web app.
  4. Atur Execute as: Me, dan Who has access: Anyone, lalu klik Deploy.
  5. Salin Web app URL yang muncul, tempel di kolom di bawah, lalu klik SIMPAN & TES.

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Waktu", "Nama Siswa", "Kelas", "Nama File", "Link File Google Drive", "Status"]);
  }
  var data = JSON.parse(e.postData.contents);
  var folder = DriveApp.getRootFolder();
  var fileUrls = [];
  if (data.files && data.files.length > 0) {
    data.files.forEach(function(f) {
      var decoded = Utilities.base64Decode(f.base64);
      var blob = Utilities.newBlob(decoded, f.type || "application/octet-stream", data.studentName + "_" + data.grade + "_" + f.name);
      var savedFile = folder.createFile(blob);
      fileUrls.push(savedFile.getUrl());
    });
  }
  sheet.appendRow([new Date(), data.studentName, data.grade, data.fileNames, fileUrls.join(", "), "Terkumpul"]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var records = [];
  for (var i = rows.length - 1; i >= 1; i--) {
    records.push({
      timestamp: Utilities.formatDate(new Date(rows[i][0]), Session.getScriptTimeZone(), "dd MMM yyyy, HH:mm"),
      studentName: rows[i][1],
      grade: rows[i][2],
      fileNames: rows[i][3],
      fileUrl: rows[i][4] || "",
      status: rows[i][5] || "Terkumpul ✓"
    });
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success", records: records})).setMimeType(ContentService.MimeType.JSON);
}
